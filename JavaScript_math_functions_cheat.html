Rounding Methods


console.log(Math.round(4.5));   // 5   → rounds to nearest integer
console.log(Math.round(4.4));   // 4

console.log(Math.ceil(4.1));    // 5   → rounds UP
console.log(Math.floor(4.9));   // 4   → rounds DOWN

console.log(Math.trunc(4.9));   // 4   → removes decimal part (no rounding)
console.log(Math.trunc(-4.9));  // -4

Min / Max


        console.log(Math.max(3, 7, 1, 9)); // 9  → largest value
console.log(Math.min(3, 7, 1, 9)); // 1  → smallest value

// With an array, use spread
let nums = [4, 8, 2, 6];
console.log(Math.max(...nums)); // 8

Power & Roots


console.log(Math.pow(2, 3));   // 8   → 2 to the power of 3
console.log(2 ** 3);           // 8   → same thing, modern syntax

console.log(Math.sqrt(16));    // 4   → square root
console.log(Math.cbrt(27));    // 3   → cube root

Absolute Value


console.log(Math.abs(-5));  // 5
console.log(Math.abs(5));   // 5

Random numbers


console.log(Math.random()); // random decimal between 0 (inclusive) and 1 (exclusive)

// Random integer between 1 and 10
let randomNum = Math.floor(Math.random() * 10) + 1;
console.log(randomNum);

// Random integer in a range (min to max, inclusive)
function randomInRange(min, max) {
  return Math.floor(Math.random() * (max - min + 1)) + min;
}
console.log(randomInRange(5, 15));

Trigonometry


    console.log(Math.sin(Math.PI / 2)); // 1
console.log(Math.cos(0));           // 1
console.log(Math.tan(0));           // 0

Logarithms


console.log(Math.log(1));    // 0    → natural log (base e)
console.log(Math.log2(8));   // 3    → log base 2
console.log(Math.log10(100)); // 2   → log base 10
