JavaScript loops

The loop below logs to console the numbers from 0 to 4.


        for (let i = 0; i < 5; i++) {
            console.log(i)
        }

i++ means adding 1

You can also do something else


        for (let i = 0; i < 5; i*= 2) {
            console.log(i)
        }

basic syntaxis


        for (initialization; condition; increment) {
            code to run
        }
    

for (let i = 0; i < 7; i++) {
    if (i%2 === 0) {
        console.log("Even");
    }
    else {
        console.log("Odd");
    }
}

The while loop


        while (condition) {
            do something
        }

Specific example


let count = 0;
while (count < 3) {
  console.log("Count: " + count);
  count++;
}

Password validation using the while loop


const password = "1234";
let inputPassw = "";

while (password != inputPassw) {
  inputPassw = prompt("Input password");
}

do while loop

This loop is going to do action one time and then evaluate the condition. 


let num = 0;
do {
  console.log("Number: " + num);
  num++;
} while (num < 3);

forEach() loop

This loop runs through an array and applies the function at each iteration. 


        array.forEach(function(item, index) {
            // do something;
        })

for of loop


let fruits = ["apple", "banana", "cherry"];
    for (let fruit of fruits) {
        console.log(fruit);
}

  let text = "";
for (let i = 0; i < 10; i++) {
  if (i === 3) { break; }
  text += "The number is " + i + "
";
}
    console.log(text);
