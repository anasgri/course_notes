        async function fetchData() {
  try {
    let response = await fetch("https://api.sampleapis.com/coffee/hot");
    if (!response.ok) {
      throw new Error(
        `Something not right with the response ${response.status}`,
      );
    }

    let data = await response.json();
    console.log(data);
  } catch (error) {
    console.log(error);
  }
}

fetchData();
